const MARKET_BASE = '/accelerator/market';

const removeEmptyValues = <T extends Record<string, unknown>>(
  values: T,
): Record<string, unknown> =>
  Object.fromEntries(
    Object.entries(values).filter(
      ([, value]) =>
        value !== undefined &&
        value !== null &&
        value !== '',
    ),
  );

export const getCareerPaths = async (
  query: MarketPathQuery = {},
): Promise<MarketPathCard[]> => {
  const response = await apiClient.get<ApiResponse<MarketPathCard[]>>(
    `${MARKET_BASE}/career-paths`,
    {
      params: removeEmptyValues({
        search: query.search,
        category: query.category,
        location: query.location,
        experience: query.experience,
        sort: query.sort,
        page: query.page ?? 1,
        pageSize: query.pageSize ?? 20,
      }),
    },
  );

  return response.data.data;
};

export const getCareerPath = async (

type PageImport = Promise<Record<string, ComponentType>>;

const lazyPage = (loadPage: () => PageImport, exportName: string) => {
  const LazyPage: LazyExoticComponent<ComponentType> = lazy(async () => {
    const module = await loadPage();

    const Page = module[exportName];

    if (!Page) throw new Error(`Lazy page export "${exportName}" was not found.`);

    return { default: Page };
  });

  return (
    <Suspense fallback={<RouteLoadingFallback />}>
      <LazyPage />
    </Suspense>
  );
};
