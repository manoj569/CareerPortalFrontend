import {
  Chip,
  Collapse,
  List,
  ListItemButton,
  ListItemIcon,
  ListItemText,
  Tooltip,
} from '@mui/material';
import ExpandLessIcon from '@mui/icons-material/ExpandLess';
import ExpandMoreIcon from '@mui/icons-material/ExpandMore';
import { forwardRef, useEffect, useId, useState } from 'react';
import { matchPath, NavLink, useLocation, type NavLinkProps } from 'react-router-dom';
import {
  SHOW_RESUME_CHECKER_NAVIGATION,
  SHOW_RESUME_MAKER_NAVIGATION,
} from '@/constants/navigationVisibility';
import { ROUTES } from '@/constants/routes';
import type { NavigationItem } from './navigationTypes';

interface NavigationListProps {
  ariaLabel?: string;
  items: NavigationItem[];
  onNavigate?: () => void;
  onAction?: (action: string) => void;
  compact?: boolean;
}
// NavLink's pathname-only active class would also highlight ordinary jobs in referral mode.
const NavigationLink = forwardRef<HTMLAnchorElement, NavLinkProps>((props, ref) => (
  <NavLink
    {...props}
    ref={ref}
    className={() => (typeof props.className === 'string' ? props.className : '')}
  />
));
const visible = (item: NavigationItem) =>
  (SHOW_RESUME_CHECKER_NAVIGATION || item.label.trim().toLowerCase() !== 'resume checker') &&
  (SHOW_RESUME_MAKER_NAVIGATION || item.path !== ROUTES.RESUME_MAKER);
const isActive = (item: NavigationItem, pathname: string, search: string): boolean => {
  if (!item.path) return false;
  const [path, query] = item.path.split('?');
  if (item.activePaths?.some((pattern) => matchPath({ path: pattern, end: true }, pathname)))
    return true;
  if (!matchPath({ path: path!, end: item.end ?? false }, pathname)) return false;
  const params = new URLSearchParams(search);
  if (query)
    return [...new URLSearchParams(query)].every(([key, value]) => params.get(key) === value);
  // The regular jobs entry and referral jobs share a pathname.
  if (item.path === '/dashboard/jobs' && params.get('mode') === 'referral') return false;
  return true;
};
const itemSx = (compact: boolean) => ({
  borderRadius: 2,
  justifyContent: compact ? 'center' : 'flex-start',
  mb: 0.5,
  minHeight: compact ? 46 : 48,
  px: compact ? 1 : 1.75,
  '&.active': {
    bgcolor: 'primary.main',
    color: 'primary.contrastText',
    '& .MuiListItemIcon-root': { color: 'inherit' },
    '&:hover': { bgcolor: 'primary.dark' },
  },
});
const NavigationEntry = ({
  item,
  onNavigate,
  onAction,
  compact = false,
}: {
  item: NavigationItem;
  onNavigate?: () => void;
  onAction?: (action: string) => void;
  compact?: boolean;
}) => {
  const location = useLocation();
  const children = item.children?.filter(visible);
  const active =
    children?.some((child) => isActive(child, location.pathname, location.search)) ??
    isActive(item, location.pathname, location.search);
  const [expanded, setExpanded] = useState(active);
  const childrenId = useId();
  useEffect(() => {
    if (active && children?.length) setExpanded(true);
  }, [active, location.pathname, location.search, children?.length]);
  const Icon = item.icon;
  const content = (
    <>
      {Icon && (
        <ListItemIcon
          sx={{ color: 'text.secondary', justifyContent: 'center', minWidth: compact ? 0 : 42 }}
        >
          <Icon sx={{ fontSize: 22 }} />
        </ListItemIcon>
      )}
      {!compact && (
        <ListItemText
          primary={item.label}
          primaryTypographyProps={{ fontWeight: 650, fontSize: '0.9rem' }}
          secondary={item.secondaryLabel}
          secondaryTypographyProps={{ variant: 'caption' }}
        />
      )}
      {!compact && item.badge && (
        <Chip
          label={item.badge}
          size="small"
          color="primary"
          sx={{ height: 20, fontSize: 11, fontWeight: 700 }}
        />
      )}
      {children?.length ? (
        expanded ? (
          <ExpandLessIcon fontSize="small" />
        ) : (
          <ExpandMoreIcon fontSize="small" />
        )
      ) : null}
    </>
  );
  return (
    <>
      <Tooltip placement="right" title={compact ? item.label : ''}>
        {children?.length ? (
          <ListItemButton
            component="button"
            type="button"
            aria-label={item.label}
            aria-expanded={expanded}
            aria-controls={childrenId}
            onClick={() => setExpanded((value) => !value)}
            sx={{ ...itemSx(compact), width: '100%', color: active ? 'primary.main' : undefined }}
          >
            {content}
          </ListItemButton>
        ) : item.action ? (
          <ListItemButton
            aria-label={item.secondaryLabel ? `${item.label}, ${item.secondaryLabel}` : item.label}
            component="button"
            onClick={() => {
              onAction?.(item.action!);
              onNavigate?.();
            }}
            sx={{ ...itemSx(compact), color: 'text.secondary', width: '100%' }}
            type="button"
          >
            {content}
          </ListItemButton>
        ) : (
          <ListItemButton
            aria-label={compact ? item.label : undefined}
            aria-current={active ? 'page' : false}
            component={NavigationLink}
            className={active ? 'active' : ''}
            end={item.end}
            onClick={onNavigate}
            sx={itemSx(compact)}
            state={
              item.preserveCurrentLocation
                ? { from: `${location.pathname}${location.search}${location.hash}` }
                : undefined
            }
            to={item.path ?? '#'}
          >
            {content}
          </ListItemButton>
        )}
      </Tooltip>
      {children?.length ? (
        <Collapse in={expanded} timeout="auto" unmountOnExit>
          <List
            id={childrenId}
            aria-label={`${item.label} navigation`}
            disablePadding
            sx={{ pl: compact ? 0 : 2 }}
          >
            {children.map((child) => (
              <NavigationEntry
                key={child.path}
                item={child}
                onNavigate={onNavigate}
                onAction={onAction}
                compact={compact}
              />
            ))}
          </List>
        </Collapse>
      ) : null}
    </>
  );
};
export const NavigationList = ({
  ariaLabel = 'Primary navigation',
  items,
  onNavigate,
  onAction,
  compact = false,
}: NavigationListProps) => (
  <List component="nav" aria-label={ariaLabel} sx={{ px: compact ? 0.75 : 1 }}>
    {items.filter(visible).map((item) => (
      <NavigationEntry key={item.path ?? item.action ?? item.label} item={item} onNavigate={onNavigate} onAction={onAction} compact={compact} />
    ))}
  </List>
);

// Refresh may happen immediately after a Render cold start.
// Give authentication refresh more time than normal API requests.
const REFRESH_TIMEOUT_MS = Math.max(environment.apiTimeoutMs, 30_000);

const refreshClient: AxiosInstance = axios.create({
  baseURL: environment.apiBaseUrl,
  timeout: REFRESH_TIMEOUT_MS,
  headers: { Accept: 'application/json' },

let refreshPromise: Promise<AuthenticationResponse> | null = null;

const endpointsWithoutAutomaticRefresh = new Set<string>([
  API_ENDPOINTS.AUTH.REGISTER,
  API_ENDPOINTS.AUTH.VERIFY_REGISTRATION_OTP,
  API_ENDPOINTS.AUTH.RESEND_REGISTRATION_OTP,
  API_ENDPOINTS.AUTH.LOGIN,
  API_ENDPOINTS.AUTH.REQUEST_LOGIN_OTP,
  API_ENDPOINTS.AUTH.LOGIN_WITH_OTP,
  API_ENDPOINTS.AUTH.REFRESH_TOKEN,
  API_ENDPOINTS.AUTH.REQUEST_PASSWORD_RESET,
  API_ENDPOINTS.AUTH.COMPLETE_PASSWORD_RESET,

const refreshAccessToken = async (): Promise<AuthenticationResponse> => {
  const refreshToken = authTokenStore.getRefreshToken();

  if (!refreshToken) {
    throw new Error('No refresh token is available.');
  }

  const response = await refreshClient.post<AuthenticationResponse>(
    API_ENDPOINTS.AUTH.REFRESH_TOKEN,
    { refreshToken },
  );

  authTokenStore.set(response.data);

  window.dispatchEvent(
    new CustomEvent('auth:session-refreshed', {
      detail: {
        accessTokenExpiresAtUtc: response.data.accessTokenExpiresAtUtc,
        user: response.data.user,
      },
    }),
  );

  return response.data;
};

const shouldExpireSession = (error: unknown): boolean => {
  if (!axios.isAxiosError(error)) {
    // No refresh token / unexpected local authentication failure.
    return true;
  }

  const status = error.response?.status;

  // These mean the refresh token itself is rejected/invalid.
  return status === 400 || status === 401 || status === 403;
};

apiClient.interceptors.request.use((config) => {
  const accessToken = authTokenStore.getAccessToken();

  if (accessToken) {
    config.headers.Authorization = `Bearer ${accessToken}`;
  }

  return config;
});

apiClient.interceptors.response.use(
  (response) => response,

  async (error: AxiosError) => {
    const originalRequest =
      error.config as RetriableRequestConfig | undefined;

    const skipsAutomaticRefresh = originalRequest?.url
      ? endpointsWithoutAutomaticRefresh.has(originalRequest.url)
      : false;

    if (
      error.response?.status !== 401 ||
      !originalRequest ||
      originalRequest.hasRetried ||
      skipsAutomaticRefresh
    ) {
      return Promise.reject(toApiRequestError(error));
    }

    originalRequest.hasRetried = true;

    try {
      // Multiple simultaneous 401 responses share one refresh request.
      refreshPromise ??= refreshAccessToken().finally(() => {
        refreshPromise = null;
      });

      const response = await refreshPromise;

      originalRequest.headers.Authorization =
        `Bearer ${response.accessToken}`;

      return apiClient(originalRequest);
    } catch (refreshError: unknown) {
      // Important:
      // Do NOT destroy the user's session because Render/network temporarily
      // timed out. Only expire when the refresh token is actually rejected.
      if (shouldExpireSession(refreshError)) {
        authTokenStore.clear();

        window.dispatchEvent(
          new CustomEvent('auth:session-expired'),
        );
      }

      return Promise.reject(
        toApiRequestError(refreshError),
      );
    }
  },
);

export { apiClient };

import type { AxiosRequestConfig, Method, RawAxiosRequestHeaders, ResponseType } from 'axios';
import type { BaseQueryApi } from '@reduxjs/toolkit/query';

import { apiClient } from './apiClient';
import { toApiRequestError } from './apiError';

export interface AxiosBaseQueryError {
  status?: number;
  message: string;
  code?: string;
  errors?: Record<string, string[]>;
  retryAfterSeconds?: number;
  redirectToMembership?: boolean;
}

interface AxiosBaseQueryArgs {
  url: string;
  method?: Method;
  data?: unknown;
  headers?: RawAxiosRequestHeaders;
  params?: object;
  responseType?: ResponseType;
  timeout?: number;
  validateStatus?: AxiosRequestConfig['validateStatus'];
}

export const axiosBaseQuery =
  () =>
  async (
    {
      data,
      headers,
      method = 'GET',
      params,
      responseType,
      timeout,
      url,
      validateStatus,
    }: AxiosBaseQueryArgs,
    { signal }: Pick<BaseQueryApi, 'dispatch' | 'signal'>,
  ) => {
    try {
      const response = await apiClient.request({
        data,
        headers,
        method,
        params,
        responseType,
        signal,
        url,
        ...(validateStatus === undefined ? {} : { validateStatus }),
        ...(timeout === undefined ? {} : { timeout }),
      });
      return { data: response.data as unknown };
    } catch (error: unknown) {
      const apiError = toApiRequestError(error);
      return {
        error: {
          code: apiError.code,
          errors: apiError.errors,
          message: apiError.message,
          redirectToMembership: apiError.redirectToMembership,
          retryAfterSeconds: apiError.retryAfterSeconds,
          status: apiError.status,
        } satisfies AxiosBaseQueryError,
      };
    }
  };
